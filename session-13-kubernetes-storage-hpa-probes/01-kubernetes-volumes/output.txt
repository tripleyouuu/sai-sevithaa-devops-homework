$ kubectl apply -f emptydir-pod.yaml -f hostpath-pod.yaml
pod/emptydir-demo created
pod/hostpath-demo created

$ kubectl exec emptydir-demo -- sh -c 'echo hello > /data/file.txt && cat /data/file.txt'
hello

$ kubectl exec hostpath-demo -- sh -c 'echo world > /data/file.txt && cat /data/file.txt'
world

$ minikube ssh -- cat /tmp/hostpath-data/file.txt   (confirm it really landed on the node, not just in the container)
world

pod "emptydir-demo" deleted from default namespace
pod/emptydir-demo created
